(function(){var A=AX,esc=A.esc,ST=[],f='problems';
function merge(d){var m={};['metar','taf'].forEach(function(k){(d.wx[k].stations||[]).forEach(function(s){var o=m[s.id]=m[s.id]||{id:s.id};o[k]=s});(d.wx[k].missing||[]).forEach(function(id){var o=m[id]=m[id]||{id:id};o[k+'Miss']=1})});
  return Object.keys(m).sort().map(function(id){var o=m[id],now=Date.now(),mt=o.metar,tf=o.taf;o.mAge=mt&&mt.site?now-mt.site:NaN;o.tAge=tf&&tf.site?now-tf.site:NaN;
    o.behind=(mt&&mt.up>mt.site&&mt.site)||(tf&&tf.up>tf.site&&tf.site);o.stale=!(o.mAge<90*6e4);o.problem=o.metarMiss||o.tafMiss||o.behind||(mt&&!mt.site);return o})}
function drawWx(){var q=A.$('#q').value.trim().toUpperCase(),L=ST.filter(function(o){return(f==='all'||(f==='stale'?o.stale:o.problem))&&(!q||o.id.indexOf(q)>=0)});

    return'<div class="ws"><b>'+esc(o.id)+'</b><span>METAR '+(o.metar&&o.metar.site?A.dur(o.mAge)+' old':'—')+' · TAF '+(o.taf&&o.taf.site?A.dur(o.tAge)+' old':'—')+'</span>'+st+(o.metar?'<span class="raw">'+esc(o.metar.raw)+'</span>':'')+'</div>'}).join('')+'</div>';
  if(!L.length)h='<p class="m">'+(f==='problems'?'No problems: every station matches the official source.':'Nothing matches.')+'</p>';
  var b=A.$('#c-wx .b');b.className='b';b.innerHTML=h}
document.querySelectorAll('#wx-seg button').forEach(function(b){b.onclick=function(){document.querySelectorAll('#wx-seg button').forEach(function(x){x.classList.toggle('on',x===b)});f=b.dataset.f;drawWx()}});
A.$('#q').addEventListener('input',drawWx);
